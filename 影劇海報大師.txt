// --- Helper Functions ---

const loadImage = (src) => {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.crossOrigin = "anonymous";
    img.onload = () => resolve(img);
    img.onerror = (e) => reject(e);
    img.src = src;
  });
};

// --- API Helpers ---

const generateTheme = async (apiKey, role, partnerName, language, userName, genre, dramaStyleId) => {
  const finalKey = apiKey || ""; 
  
  const langMap = {
    'zh-TW': 'Traditional Chinese (繁體中文)',
    'en': 'English',
    'ja': 'Japanese',
    'ko': 'Korean'
  };
  
  const styleConfig = DRAMA_STYLES.find(s => s.id === dramaStyleId) || DRAMA_STYLES[0];
  const targetLang = langMap[language] || 'Traditional Chinese';
  const userDisplayName = userName || "The User";
  const selectedGenre = genre !== "random" ? genre : "Any suitable genre";
  const dramaPrompt = styleConfig.prompt;
  const titleLangCode = styleConfig.titleLang;

  const prompt = `Generate a catchy title and a short plot summary for a new ${dramaPrompt} poster. 
  Leads: ${userDisplayName} (${role}), ${partnerName}. 
  Target Language for Plot: ${targetLang}.
  Specific Genre: ${selectedGenre}.
  Country of Origin: ${styleConfig.country}.
  
  Instructions:
  1. Title should be catchy and fit the genre "${selectedGenre}" and style "${dramaPrompt}".
  2. Plot should be in ${targetLang} and strictly follow the genre tropes of ${dramaPrompt}.
  3. "nativeTitle" should be in the native language of the drama style (e.g., Korean for K-Drama, Japanese for J-Drama, Chinese for C-Drama, English for US Series).
  4. Genre determines the mood, lighting, and costume design in the visual description.
  5. Incorporate the lead name "${userDisplayName}" into the plot if it fits naturally.
  
  Return JSON: { 
    "title": "Title in ${targetLang} (Translated)", 
    "nativeTitle": "Original Title in ${titleLangCode}", 
    "genre": "Should match ${selectedGenre}", 
    "fontStyle": "elegant|heavy|cute|handwritten|modern|retro", 
    "plot": "Summary in ${targetLang}", 
    "visualPrompt": "Detailed visual description in English for image generation, focusing on the mood of ${dramaPrompt} and genre ${selectedGenre}. Mention specific cultural clothing or setting styles if applicable (e.g. Hanbok for Historical K-Drama, Kimono/Office for J-Drama, Wuxia robes for C-Drama)." 
  }`;

  const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${finalKey}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      contents: [{ parts: [{ text: prompt }] }],
      generationConfig: { responseMimeType: "application/json" }
    })
  });
  const data = await response.json();
  return JSON.parse(data.candidates[0].content.parts[0].text);
};

const generateImage = async (userKey, base64Image, theme, role, partnerName, aspectRatio, resolution, userName, dramaStyleId) => {
  const apiKey = userKey || "";
  // Always use Nano Banana (Gemini 2.5 Flash Image Preview) as it's the supported model for this environment
  const model = "gemini-2.5-flash-image-preview";
  const userDisplayName = userName || "Main Character";
  const styleConfig = DRAMA_STYLES.find(s => s.id === dramaStyleId) || DRAMA_STYLES[0];

  const prompt = `A cinematic ${styleConfig.prompt} poster for genre "${theme.genre}". 
  Title text to include: "${theme.nativeTitle}". 
  Credits text to include (optional): "${userDisplayName}" & "${partnerName}".
  Lead: user-face (representing ${userDisplayName}) + ${partnerName}. 
  Scene: ${theme.visualPrompt}. 
  Style: ${styleConfig.country} TV Drama aesthetic.
  Target Aspect Ratio: ${aspectRatio}.
  Professional photography, movie poster quality, high details.`;

  const payload = {
    contents: [{ parts: [{ text: prompt }, { inlineData: { mimeType: "image/jpeg", data: base64Image } }] }],
    generationConfig: {
      responseModalities: ['TEXT', 'IMAGE']
    }
  };

  const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${apiKey}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload)
  });
  const data = await response.json();
  const part = data.candidates?.[0]?.content?.parts?.find(p => p.inlineData);
  if (part) return `data:image/jpeg;base64,${part.inlineData.data}`;
  throw new Error("Failed to generate image.");
};

      {children}
    </button>
  );
};

  const [userImage, setUserImage] = useState(null);
  const [role, setRole] = useState("male");
  const [dramaStyle, setDramaStyle] = useState('k-drama'); 
  const [userName, setUserName] = useState(""); 
  const [selectedGenre, setSelectedGenre] = useState("random");
  const [aspectRatio, setAspectRatio] = useState("2:3");
  const [resolution, setResolution] = useState("2K");
  const [isGenerating, setIsGenerating] = useState(false);
  const [isRegenerating, setIsRegenerating] = useState(false);
  const [generatedResult, setGeneratedResult] = useState(null);
  const [error, setError] = useState(null);
  const [language, setLanguage] = useState("zh-TW");
  
  const fileInputRef = useRef(null);
  const t = TRANSLATIONS[language] || TRANSLATIONS['zh-TW'];

  useEffect(() => {
    setSelectedGenre("random");
  }, [dramaStyle]);

  useEffect(() => {
    fetch(`https://callmygod.com/api/game_check.php?game=${APP_ID}`).catch(err => console.error(err));
  }, []);

  const handleFileUpload = (e) => {
    const file = e.target.files[0];
    if (file) {
      const reader = new FileReader();
      reader.onloadend = () => setUserImage(reader.result);
      reader.readAsDataURL(file);
    }
  };

  const handleGenerate = async () => {
    if (!userImage) { setError(t.errorUpload); return; }

    setIsGenerating(true);
    setError(null);
    try {
      const styleKey = dramaStyle;
      const actorList = role === 'male' ? ACTOR_DB[styleKey].women : ACTOR_DB[styleKey].men;
      const partner = actorList[Math.floor(Math.random() * actorList.length)];
      
      const theme = await generateTheme(apiKey, role, partner, language, userName, selectedGenre, dramaStyle);
      const rawBase64 = userImage.split(',')[1]; 
      // Default to Nano Banana (gemini-2.5-flash-image-preview)
      const imageUrl = await generateImage(apiKey, rawBase64, theme, role, partner, aspectRatio, resolution, userName, dramaStyle);
      
      setGeneratedResult({ 
        ...theme, 
        imageUrl, 
        partner, 
        userName,
        styleId: dramaStyle
      });
    } catch (err) {
      setError(err.message);
    } finally {
      setIsGenerating(false);
    }
  };

  const handleRegenerateImage = async () => {
    if (!generatedResult || !userImage) return;
    setIsRegenerating(true);
    setError(null);
    try {
      const rawBase64 = userImage.split(',')[1];
      const imageUrl = await generateImage(
        apiKey, 
        rawBase64, 
        generatedResult, 
        role, 
        generatedResult.partner, 
        aspectRatio, 
        resolution,
        userName,
        generatedResult.styleId
      );
      
      setGeneratedResult(prev => ({ ...prev, imageUrl }));
    } catch (err) {
      setError(err.message);
    } finally {
      setIsRegenerating(false);
    }
  };

  const handleDownload = async () => {
    if (!generatedResult?.imageUrl) return;

    try {
      const canvas = document.createElement('canvas');
      const ctx = canvas.getContext('2d');

      const mainImg = await loadImage(generatedResult.imageUrl);
      canvas.width = mainImg.width;
      canvas.height = mainImg.height;
      
      ctx.drawImage(mainImg, 0, 0);

      const randomNum = Math.floor(10000000 + Math.random() * 90000000);
      const timestamp = Date.now();
      const filename = `DramaPoster_${timestamp}_${randomNum}.jpg`;

      const dataUrl = canvas.toDataURL('image/jpeg', 0.92);
      
      const link = document.createElement('a');
      link.href = dataUrl;
      link.download = filename;
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    } catch (err) {
      console.error("Download failed:", err);
      setError("Download failed. Please try again.");
    }
  };

  const fontStyleObj = generatedResult ? (FONT_STYLES[generatedResult.fontStyle] || FONT_STYLES.elegant) : FONT_STYLES.elegant;
  const currentGenres = GENRE_DB[dramaStyle] || GENRE_DB['k-drama'];

  return (
    <div className="min-h-screen bg-[#DDF4DE] font-sans pb-10">
      <style>{`@import url('https://fonts.googleapis.com/css2?family=Black+Han+Sans&family=Do+Hyeon&family=Gamja+Flower&family=Nanum+Brush+Script&family=Nanum+Myeongjo:wght@400;700&family=Song+Myung&display=swap');`}</style>

      {/* Navbar */}
      <nav className="bg-[#88DDBB] p-4 shadow-md sticky top-0 z-40 border-b-4 border-white/30">
        <div className="max-w-4xl mx-auto flex flex-col md:flex-row justify-between items-center gap-4 md:gap-0">
          <div className="flex items-center gap-3">
            <Film className="text-[#FF8AA3] w-6 h-6 fill-current" />
            <h1 className="text-white text-xl md:text-2xl font-black">{t.title}</h1>
          </div>
          <div className="flex items-center gap-3 flex-wrap justify-center">
            {/* Language Switcher */}
            <div className="flex bg-black/10 rounded-full p-1 border border-white/20">
              {['zh-TW', 'en', 'ja', 'ko'].map(lang => (
                <button 
                  key={lang}
                  onClick={() => setLanguage(lang)} 
                  className={`w-8 h-8 rounded-full text-xs font-black transition-all flex items-center justify-center ${language === lang ? "bg-white text-[#7FD1B9]" : "text-white opacity-60 hover:opacity-100"}`}
                  title={TRANSLATIONS[lang].langName}
                >
                  {lang === 'zh-TW' ? '繁' : lang === 'en' ? 'EN' : lang === 'ja' ? 'JP' : 'KR'}
                </button>
              ))}
            </div>

            <button onClick={() => setShowKeyModal(true)} className="bg-white/90 p-2 rounded-full text-[#5A4D3B]"><Settings className="w-5 h-5" /></button>
          </div>
        </div>
      </nav>

      <div className="max-w-4xl mx-auto p-4 md:p-8 space-y-8">
        <div className="grid grid-cols-1 md:grid-cols-2 gap-8">
          <div className="space-y-6">
            <Card>
              <h2 className="text-xl font-black text-[#5A4D3B] mb-4">1. {t.uploadTitle}</h2>
              <div onClick={() => fileInputRef.current?.click()} className={`border-4 border-dashed rounded-3xl h-64 flex flex-col items-center justify-center cursor-pointer overflow-hidden ${userImage ? 'border-[#7FD1B9]' : 'border-[#E0D5C1]'}`}>
                {userImage ? <img src={userImage} className="w-full h-full object-cover" /> : <Upload className="w-10 h-10 text-[#E0D5C1]" />}
                <input type="file" ref={fileInputRef} onChange={handleFileUpload} accept="image/*" className="hidden" />
              </div>
            </Card>
            <Card>
              <h2 className="text-xl font-black text-[#5A4D3B] mb-4">2. {t.settingsTitle}</h2>
              <div className="space-y-4">
                
                {/* Drama Style Selector */}
                <div>
                  <label className="text-xs font-bold text-[#8B7E66] ml-2 flex items-center gap-1"><Globe className="w-3 h-3"/> {t.styleLabel}</label>
                  <select 
                    value={dramaStyle} 
                    onChange={(e) => setDramaStyle(e.target.value)}
                    className="w-full bg-white border-2 border-[#E0D5C1] rounded-xl p-3 font-bold text-[#5A4D3B]"
                  >
                    {DRAMA_STYLES.map(opt => (
                      <option key={opt.id} value={opt.id}>{opt.label}</option>
                    ))}
                  </select>
                </div>

                <div className="flex bg-white rounded-2xl border-2 border-[#E0D5C1] p-1">
                  <button onClick={() => setRole('male')} className={`flex-1 py-2 rounded-xl font-bold ${role === 'male' ? 'bg-[#5E97F6] text-white' : 'text-[#8B7E66]'}`}>{t.maleLead}</button>
                  <button onClick={() => setRole('female')} className={`flex-1 py-2 rounded-xl font-bold ${role === 'female' ? 'bg-[#FF8AA3] text-white' : 'text-[#8B7E66]'}`}>{t.femaleLead}</button>
                </div>

                {/* Name Input Field */}
                <div>
                  <label className="text-xs font-bold text-[#8B7E66] ml-2 flex items-center gap-1"><User className="w-3 h-3"/> {t.nameLabel}</label>
                  <input 
                    type="text" 
                    value={userName} 
                    onChange={(e) => setUserName(e.target.value)}
                    placeholder={t.namePlaceholder}
                    className="w-full bg-white border-2 border-[#E0D5C1] rounded-xl p-3 font-bold text-[#5A4D3B] placeholder-[#CAB08B]"
                  />
                </div>

                {/* Genre Selection Field */}
                <div>
                  <label className="text-xs font-bold text-[#8B7E66] ml-2 flex items-center gap-1"><Clapperboard className="w-3 h-3"/> {t.genreLabel}</label>
                  <select 
                    value={selectedGenre} 
                    onChange={(e) => setSelectedGenre(e.target.value)}
                    className="w-full bg-white border-2 border-[#E0D5C1] rounded-xl p-3 font-bold text-[#5A4D3B]"
                  >
                    {currentGenres.map(opt => (
                      <option key={opt.value} value={opt.value}>{opt.label}</option>
                    ))}
                  </select>
                </div>

                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <label className="text-xs font-bold text-[#8B7E66] ml-2">{t.aspectRatio}</label>
                    <select value={aspectRatio} onChange={(e)=>setAspectRatio(e.target.value)} className="w-full bg-white border-2 border-[#E0D5C1] rounded-xl p-2 font-bold text-[#5A4D3B]">
                      {["1:1","2:3","3:2","3:4","4:3","4:5","5:4","9:16","16:9","21:9"].map(r => <option key={r} value={r}>{r}</option>)}
                    </select>
                  </div>
                  <div>
                    <label className="text-xs font-bold text-[#8B7E66] ml-2">{t.quality}</label>
                    <select value={resolution} onChange={(e)=>setResolution(e.target.value)} className="w-full bg-white border-2 border-[#E0D5C1] rounded-xl p-2 font-bold text-[#5A4D3B]">
                      {["2K", "4K"].map(r => <option key={r} value={r}>{r}</option>)}
                    </select>
                  </div>
                </div>
                <Button onClick={handleGenerate} disabled={isGenerating || isRegenerating} className="w-full">
                  {isGenerating ? <RefreshCw className="animate-spin" /> : <Sparkles />} {t.generateBtn}
                </Button>
                {error && <div className="text-red-500 text-xs font-bold text-center">{error}</div>}
              </div>
            </Card>
          </div>

          <Card className="min-h-[500px] flex flex-col">
            <h2 className="text-xl font-black text-[#5A4D3B] mb-4">3. {t.resultTitle}</h2>
            <div className="flex-1 relative bg-white rounded-3xl border-2 border-[#E0D5C1] flex items-center justify-center overflow-hidden group">
              {generatedResult ? (
                <>
                  <img src={generatedResult.imageUrl} className="w-full h-full object-contain" />
                  
                  {/* Overlay Actions */}
                  {!isRegenerating && (
                    <div className="absolute top-4 right-4 flex flex-col gap-2 opacity-0 group-hover:opacity-100 transition-opacity">
                      <Button variant="icon" onClick={handleRegenerateImage} title="Regenerate Image">
                        <RefreshCw className="w-5 h-5 text-[#5E97F6]" />
                      </Button>
                      <Button variant="icon" onClick={handleDownload} title="Download">
                        <Download className="w-5 h-5 text-[#7FD1B9]" />
                      </Button>
                    </div>
                  )}

                  {/* Regenerating Overlay */}
                  {isRegenerating && (
                    <div className="absolute inset-0 bg-white/50 backdrop-blur-sm flex items-center justify-center">
                       <div className="bg-white px-4 py-2 rounded-full shadow-lg flex items-center gap-2">
                         <RefreshCw className="animate-spin text-[#5E97F6] w-4 h-4" />
                         <span className="text-xs font-bold text-[#5A4D3B]">{t.regenerating}</span>
                       </div>
                    </div>
                  )}
                </>
              ) : (
                <ImageIcon className="w-20 h-20 text-[#E0D5C1] opacity-30" />
              )}
            </div>
            {generatedResult && (
              <div className="mt-4">
                <h3 style={{ fontFamily: fontStyleObj.fontFamily }} className={`text-2xl ${fontStyleObj.className} ${fontStyleObj.shadow}`}>
                  {generatedResult.nativeTitle}
                </h3>
                <p className="text-sm font-bold text-[#7FD1B9]">{generatedResult.title}</p>
                
                {/* Updated Starring Line */}
                <div className="text-[10px] font-bold text-[#8B7E66] mt-2 uppercase tracking-tighter">
                  Starring: {generatedResult.userName || t.you} & {generatedResult.partner}
                </div>
                
                <p className="text-xs text-[#8B7E66] mt-2 italic border-l-2 border-[#7FD1B9] pl-2">{generatedResult.plot}</p>
              </div>
            )}
          </Card>
        </div>
      </div>

      {/* API Key Modal */}
      {showKeyModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/40 backdrop-blur-sm">
          <Card className="w-full max-w-sm text-center">
            <h2 className="text-xl font-black mb-2">{t.apiKeyTitle}</h2>
            <p className="text-xs font-bold text-[#8B7E66] mb-4">{t.apiKeyDesc}</p>
            <input type="password" value={apiKey} onChange={(e)=>setApiKey(e.target.value)} placeholder="AIzaSy..." className="w-full border-2 border-[#E0D5C1] rounded-xl p-3 mb-4" />
            <a href="https://aistudio.google.com/api-keys" target="_blank" rel="noreferrer" className="block text-xs font-black text-[#5E97F6] mb-4 underline">{t.apiKeyLink}</a>
            <Button onClick={()=>setShowKeyModal(false)} className="w-full">{t.saveClose}</Button>
          </Card>
        </div>
      )}
    </div>
  );
}
